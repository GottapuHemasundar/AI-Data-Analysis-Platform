def load_file(uploaded_file):
    """Load file based on type (CSV, Excel, ZIP)."""
    df_list = []

    if uploaded_file.name.endswith(".zip"):
        with zipfile.ZipFile(uploaded_file) as z:
            for file in z.namelist():
                if file.endswith(".csv"):
                    df_list.append(pd.read_csv(z.open(file)))
                elif file.endswith(".xlsx"):
                    df_list.append(pd.read_excel(z.open(file)))

        if not df_list:
            st.error("ZIP file does not contain CSV or Excel files")
            return None

        df = pd.concat(df_list, ignore_index=True)
        st.success(f"ZIP uploaded successfully! {len(df_list)} file(s) merged.")

    elif uploaded_file.name.endswith(".csv"):
        df = pd.read_csv(uploaded_file)
        st.success("CSV dataset uploaded successfully!")

    else:
        df = pd.read_excel(uploaded_file)
        st.success("Excel dataset uploaded successfully!")

    return df

def clean_data(df, options):
    """Apply cleaning operations based on user options."""
    df_cleaned = df.copy()

    # Basic cleaning
    if options.get("remove_duplicates", True):
        initial_rows = len(df_cleaned)
        df_cleaned.drop_duplicates(inplace=True)
        if len(df_cleaned) < initial_rows:
            st.info(f"Removed {initial_rows - len(df_cleaned)} duplicate rows.")

    if options.get("standardize_columns", True):
        df_cleaned.columns = (
            df_cleaned.columns
            .str.strip()
            .str.lower()
            .str.replace(" ", "_")
            .str.replace("[^a-z0-9_]", "", regex=True)
        )
        st.info("Column names standardized.")

    if options.get("handle_unknowns", True):
        df_cleaned.replace("Unknown", np.nan, inplace=True)
        st.info("Converted 'Unknown' strings to NaN.")

    if options.get("trim_spaces", True):
        for col in df_cleaned.select_dtypes(include="object"):
            try:
                
                if df_cleaned[col].dtype == 'object' and df_cleaned[col].apply(lambda x: isinstance(x, str)).all():
                    df_cleaned[col] = df_cleaned[col].str.strip()
            except (AttributeError, TypeError):
           
                pass
        st.info("Trimmed spaces in text columns.")


    if options.get("fix_data_types", True):
        for col in df_cleaned.columns:
            if "date" in col.lower() or "time" in col.lower():
                df_cleaned[col] = pd.to_datetime(df_cleaned[col], errors="coerce")
            else:
                df_cleaned[col] = pd.to_numeric(df_cleaned[col], errors="ignore")
        st.info("Data types optimized.")

    if options.get("remove_negatives", True):
        num_cols = df_cleaned.select_dtypes(include="number").columns
        for col in num_cols:
            negative_count = (df_cleaned[col] < 0).sum()
            if negative_count > 0:
                df_cleaned[col] = df_cleaned[col].where(df_cleaned[col] >= 0)
                st.info(f"Removed {negative_count} negative values from {col}.")

    
    num_cols = df_cleaned.select_dtypes(include="number").columns
    cat_cols = df_cleaned.select_dtypes(include="object").columns

    fill_missing_num = options.get("fill_missing_num", "median")
    if fill_missing_num != "none" and len(num_cols) > 0:
        if fill_missing_num == "median":
            df_cleaned[num_cols] = df_cleaned[num_cols].fillna(df_cleaned[num_cols].median())
        elif fill_missing_num == "mean":
            df_cleaned[num_cols] = df_cleaned[num_cols].fillna(df_cleaned[num_cols].mean())
        elif fill_missing_num == "mode":
            for col in num_cols:
                mode_val = df_cleaned[col].mode()
                if not mode_val.empty:
                    df_cleaned[col] = df_cleaned[col].fillna(mode_val.iloc[0])
        elif fill_missing_num == "drop":
            df_cleaned = df_cleaned.dropna(subset=num_cols)
        st.info(f"Missing numeric values filled using {fill_missing_num}.")

    fill_missing_cat = options.get("fill_missing_cat", "Unknown")
    if fill_missing_cat != "none" and len(cat_cols) > 0:
        if fill_missing_cat == "Unknown":
            df_cleaned[cat_cols] = df_cleaned[cat_cols].fillna("Unknown")
        elif fill_missing_cat == "mode":
            for col in cat_cols:
                mode_val = df_cleaned[col].mode()
                if not mode_val.empty:
                    df_cleaned[col] = df_cleaned[col].fillna(mode_val.iloc[0])
        elif fill_missing_cat == "drop":
            df_cleaned = df_cleaned.dropna(subset=cat_cols)
        st.info(f"Missing categorical values filled using {fill_missing_cat}.")

    return df_cleaned

def calculate_quality(df):
    """Calculate data quality score."""
    if len(df) == 0:
        return 0

    total_cells = df.shape[0] * df.shape[1]
    missing = df.isnull().sum().sum()
    duplicates = df.duplicated().sum()

    numeric = df.select_dtypes(include="number")
    negative = (numeric < 0).sum().sum() if not numeric.empty else 0

    score = (
        (1 - missing / total_cells) * 100 +
        (1 - duplicates / len(df)) * 100 +
        (1 - negative / total_cells) * 100
    ) / 3

    return round(score, 2)
